<script>
let products = [];

function addProduct() {
    const name = document.getElementById("name").value;
    const price = document.getElementById("price").value;
    const imageInput = document.getElementById("image");

    if (!name || !price || imageInput.files.length === 0) {
        alert("Please fill all fields!");
        return;
    }

    const reader = new FileReader();
    reader.onload = function(e) {
        const product = {
            name,
            price,
            image: e.target.result,
            likes: 0,
            comments: []
        };

        products.push(product);
        displayProducts();
    };

    reader.readAsDataURL(imageInput.files[0]);
}

function displayProducts() {
    const container = document.getElementById("products");
    container.innerHTML = "";

    products.forEach((product, index) => {
        const card = document.createElement("div");
        card.className = "card";

        card.innerHTML = `
            <img src="${product.image}">
            <div class="card-content">
                <h3>${product.name}</h3>
                <p class="price">KES ${product.price}</p>

                <div class="actions">
                    <button onclick="likeProduct(${index})">❤️ ${product.likes}</button>
                </div>

                <div class="comments">
                    <input type="text" placeholder="Add comment" onkeypress="addComment(event, ${index})">
                    ${product.comments.map(c => `<div class="comment">${c}</div>`).join("")}
                </div>
            </div>
        `;

        container.appendChild(card);
    });
}

function likeProduct(index) {
    products[index].likes++;
    displayProducts();
}

function addComment(event, index) {
    if (event.key === "Enter") {
        const text = event.target.value;
        if (text.trim() !== "") {
            products[index].comments.push(text);
            displayProducts();
        }
    }
}
</script>
